275template <typename S, typename T>
276inline void Graphe<S, T>::copieDepuis(const Graphe<S, T>& other)
277{
278 _prochaineClef = other._prochaineClef;
279
280 std::unordered_map<const Sommet<T>*, Sommet<T>*> mapSommets;
281
282 // Copie des sommets
283 for (auto p = other._lSommets; p; p = p->_suivant)
284 {
285 Sommet<T>* nouveau = creeSommet(p->_info->_v);
286 nouveau->_clef = p->_info->_clef; // IMPORTANT
287 mapSommets[p->_info] = nouveau;
288 }
289
290 // Copie des arêtes
291 for (auto a = other._lAretes; a; a = a->_suivant)
292 {
293 auto oldA = a->_info;
294 Sommet<T>* s1 = mapSommets.at(oldA->_debut);
295 Sommet<T>* s2 = mapSommets.at(oldA->_fin);
296
297 creeArete(oldA->_v, s1, s2);
298 }
299}
