144inline void ServeurTCP::boucleServeur()
145{
146 log(NiveauLog::INFO, "Serveur en attente de connexions...");
147
148 while (_serveurEstActif)
149 {
150 socket_t socketClient = ::accept(_socketServeur, nullptr, nullptr);
151
152 if (!_serveurEstActif)
153 break;
154
155 if (socketClient == INVALID_SOCKET_FD)
156 {
157 log(NiveauLog::ERREUR, "accept()");
158 continue;
159 }
160
161 int idClient = ++_compteurClients;
162 log(NiveauLog::INFO, "Nouvelle connexion acceptée", idClient);
163
164 try
165 {
166 _threadsClients.emplace_back(&ServeurTCP::gererClient, this, socketClient, idClient);
167 }
168 catch (const std::exception& e)
169 {
170 log(NiveauLog::ERREUR, std::string("Impossible de créer un thread : ") + e.what(), idClient);
171 socket_close(socketClient);
172 }
173 }
174}

176inline std::string ServeurTCP::traiterRequete(const std::string& requete)
177{
178 try
179 {
180 if (!_requeteHandler)
181 return "{\"erreur\": \"Handler non initialisé\"}";
182
183 std::string commande =
184 JsonParserUtils::recupererObligatoire(requete, "commande", JsonParserUtils::extraireChampString);
185
186 std::optional<std::string> reponse = _requeteHandler->traiter(commande, requete);
187
188 return reponse.value_or("{\"erreur\": \"Aucun handler n'a pu traiter la requête\"}");
189 }
190 catch (const std::exception& e)
191 {
192 return std::string("{\"erreur\": \"") + e.what() + "\"}";
193 }
194 catch (...)
195 {
196 return "{\"erreur\": \"Erreur inconnue\"}";
197 }
198}
